\begin{recipe}
  [source = {Based on:~\cite{segalPumpkinBread2009}}]
  {Pumpkin Bread}
  \ingredients{
    2 c. & flour \\
    \nicefrac{1}{2} tsp. & salt \\
    1 tsp. & baking soda \\
    \nicefrac{1}{2} tsp. & baking powder \\
    1 tsp. & ground cloves \\
    1 tsp. & ground cinnamon \\
    1 tsp. & ground nutmeg \\
    \nicefrac{3}{4} c. & butter, softened \\
    1\nicefrac{1}{2} c. & sugar\\
    2 & eggs \\
    1 (15-oz) can  & 100\% pure pumpkin
  }

  \preparation{
    \step Preheat oven to 325\degF and set oven rack in middle positition.
    \step Grease and flow two 8 x 4-in loaf. pans.
    \step In a medium bowl, combine flour, salt, baking soda, baking powder, cloves, cinnamon, and nutmeg. Whisk until well combined and set aside.
    \step In a large bowl of an electric mixer, beat the butter and sugar on medium speed until just blended.
    \step Add the eggs one at a time, beating well after each addition.
    \step Beat until very light and fluffy (a few minutes).
    \step Beat in the pumpkin. (Mixture might look grainy -- this is expected.)
    \step Add the flour mixture gradually on low speed until just combined.
    \step. Divide batter between pans and bake for 65-75 minutes, or until toothpick inserted into the center comes out clean.
    \step Let the loaves cool in the pans for about 10 minutes, then turn onto a wire rack to cool completely.
  }

  \hint{
    \begin{itemize}
      \item Can optionally melt butter in milk in the microwave, then add to breach machine pan. Be careful to not overheat, as this could scramble the egg.
    \end{itemize}
  }
\end{recipe}
