\begin{recipe}
  [source = {Patricia Schroeder Christian~\cite{HaeckerHeritage1987}}]
  {Kathy Haecker's Kolaches}
  \ingredients{
    2 pkgs & yeast \\
    \nicefrac{1}{4} c. & warm water \\
    2 c. & milk \\
    \nicefrac{3}{4} c. & sugar \\
    \nicefrac{2}{3} c. & butter, melted \\
    2 & egg yolks, beaten \\
    2 tsp. & salt \\
    6+ c. & flour 
  }

  \preparation{
    \step Dissolve yeast in water with a teaspoon of sugar.
    \step Heat milk and butter over medium-low heat until butter is melted. Add sugar and salt and stir until dissolved.
    \step Add 2-3 cups of flour to a large mixing bowl and stir in liquid mixture.
    \step Add egg yolks and yeast mixture.
    \step Stir in flour and knead until dough holds its shape but is still soft.
    \step Cover and proof in a warm place until doubled in size.
    \step Punch down then allow to double in size again.
    \step Form and bake at 350\degF.
  }

  \hint{
    \begin{itemize}
      \item Make sure to not add the eggs and yeast to the warm mixture until cool enough to not scramble the eggs and/or kill the yeast.
      \item Baking times will vary depending on shape.
      \item A single recipe makes 2 9x13'' pans of cinnamon rolls. Bake for about 20 minutes.
    \end{itemize}
  }
\end{recipe}
