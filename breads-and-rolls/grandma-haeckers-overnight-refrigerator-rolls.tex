\begin{recipe}
  [source = {Patricia Schroeder Christian~\cite{HaeckerHeritage1987}}]
  {Grandma Haecker's Overnight Refrigerator Rolls}
  \ingredients{2 pkgs & yeast \\
  2\nicefrac{1}{2} cups & warm water \\
  \nicefrac{3}{4} cups & soft shortening \\
  \nicefrac{3}{4} cups & sugar \\
  2 & eggs, beaten \\
  8 - 8\nicefrac{1}{2} cups & flour \\
  2\nicefrac{1}{2} tsp. & salt
  }

  \preparation{
    \step Mix yeast in warm water.
    \step Add shortening, sugar, eggs, 4 cups of flour, and salt. Mix until smooth.
    \step Stir in remaining flour. This will be soft dough.
    \step Cover tightly; store in refrigerator overnight or until needed. Keeps for four days. Punch down daily.
  }

  \hint{Use vegetable oil for best rising in the refrigerator. Butter will prevent rising.}

\end{recipe}
