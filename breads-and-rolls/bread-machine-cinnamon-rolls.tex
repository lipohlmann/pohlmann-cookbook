\begin{recipe}
  [source = {Inspired by recipies in~\cite{generalmillsBestBreadMachine1999,HaeckerHeritage1987}}]
  {Bread Machine Cinnamon Rolls}
  \ingredients{
    1 c. & milk, room-temperature \\
    1 & egg \\
    \nicefrac{1}{3} c. & butter, melted \\
    \nicefrac{1}{3} c. & sugar \\
    1 tsp. & salt \\
    3\nicefrac{1}{2} c. & bread flour\\
    1.5 tsp. & quick-rise yeast
  }

  \preparation{
    \step Add all ingrediants in order to bread machine.
    \step Start bread machine on Dough cycle.
    \step Roll into a rectangle on countertop.
    \step Spread $\nicefrac{1}{4}$ cup butter, melted, onto dough and top with a cinnamon-sugar mixture.
    \step Roll tightly and cut into 1.5-inch pieces.
    \step Add to greased pans so that each roll is only just touching its neighbors (do not pack too tightly).
    \step Cover with a clean towel and place in a warm environment until doubled in size.
    \step Bake at 350\degF for 15-20 minutes.
  }

  \hint{
    \begin{itemize}
      \item Can optionally melt butter in milk in the microwave, then add to breach machine pan. Be careful to not overheat, as this could scramble the egg.
    \end{itemize}
  }
\end{recipe}
