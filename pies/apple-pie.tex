\begin{recipe}
  {Liam's Apple Pie}
  \ingredients{
    1 & pie crust \\
    4-6 cups & peeled and sliced apples \\
    \nicefrac{1}{2} c. & brown sugar \\
    \nicefrac{1}{2} c. & white sugar \\
    1 tsp. & ground cinnamon \\
    \nicefrac{1}{2} tsp. & ground nutmeg \\
    \nicefrac{1}{4} c. & all-purpose flour \\
    dash & salt \\
    2 Tbsp. & butter, cubed
  }

  \preparation{
    \step Mix together all ingredients in bowl, then pour into uncooked pie crust.
    \step Top with cubed butter.
    \step Top with second crust or crumble topping. If using a second crust, brush with milk, sprinkle with white sugar, and vent with a fork.
    \step Bake at 400\degF  for 40-50 minutes.
  }

  \hint{
    \begin{itemize}
      \item Place an old cookie sheet under pie dish when baking to avoid overflow onto the oven.
      \item Pink Lady, Fuji, Granny Smith, and Gala apples work well.
    \end{itemize}
  }
\end{recipe}
